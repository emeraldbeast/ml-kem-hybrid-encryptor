cmake_minimum_required(VERSION 3.20)
project(xwing_parallel_native VERSION 0.2.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(OpenSSL 3.5 REQUIRED COMPONENTS Crypto)
find_package(OpenMP REQUIRED COMPONENTS C)

add_library(xwing_native STATIC native/src/xwing.c)
target_include_directories(xwing_native PUBLIC native/include)
target_link_libraries(xwing_native PUBLIC OpenSSL::Crypto)
add_executable(xwing_native_demo native/src/demo.c)
target_link_libraries(xwing_native_demo PRIVATE xwing_native)
add_executable(xwing_secure_demo native/src/secure_demo.c)
target_link_libraries(xwing_secure_demo PRIVATE xwing_native OpenSSL::Crypto)
add_executable(xwing_native_bench native/src/benchmark.c)
target_link_libraries(xwing_native_bench PRIVATE xwing_native OpenMP::OpenMP_C)
add_executable(ntt_lab native/experiments/ntt/ntt_lab.c)
target_link_libraries(ntt_lab PRIVATE OpenMP::OpenMP_C)
add_executable(xwing_key_context_bench native/experiments/key_context_bench.c)
target_link_libraries(xwing_key_context_bench PRIVATE xwing_native OpenMP::OpenMP_C)
add_executable(xwing_adversarial_stress testing/stress-tests/native_adversarial.c)
target_link_libraries(xwing_adversarial_stress PRIVATE xwing_native OpenMP::OpenMP_C)

if(MSVC)
  foreach(target xwing_native xwing_native_demo xwing_secure_demo xwing_native_bench ntt_lab xwing_key_context_bench xwing_adversarial_stress)
    target_compile_options(${target} PRIVATE /W4)
  endforeach()
else()
  target_link_libraries(xwing_adversarial_stress PRIVATE m)
  foreach(target xwing_native xwing_native_demo xwing_secure_demo xwing_native_bench ntt_lab xwing_key_context_bench xwing_adversarial_stress)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endforeach()
endif()
